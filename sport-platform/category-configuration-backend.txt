import java.time.LocalDate;

@RestController
@RequestMapping("/api/v1/regulation-configurations")
@PreAuthorize("hasAnyRole('PLATFORM_ADMIN', 'ORGANIZER')")
public class UnifiedRegulationController {
    private final RegulationBundleAccess documents;
    private final ProtocolConfigurationPublication protocols;
    public UnifiedRegulationController(RegulationBundleAccess documents, ProtocolConfigurationPublication protocols) {
        this.documents = documents; this.protocols = protocols;
    }
    public record Request(UUID sourceVersionId, @NotNull UUID sportId,
            @NotBlank @Size(max=64) String code, @NotBlank @Size(max=255) String name,
            @Size(max=10000) String description, @Min(1) int versionNo, LocalDate effectiveFrom,
            @Size(max=10000) String notes,
            @NotNull @Size(max=256) List<@NotBlank @Size(max=2000) String> conditions,
            @Valid ProtocolConfigurationPublication.Settings protocol) { }
    public record Response(RegulationBundleAccess.Document document, ProtocolConfigurationPublication.Settings protocol) { }
    @PostMapping @Transactional
    public Response publish(@Valid @RequestBody Request r,
            @AuthenticationPrincipal AuthenticatedUserPrincipal principal, Authentication authentication) {
        UUID actor = actor(principal);
        boolean admin = admin(authentication);
        if (r.sourceVersionId() != null) {
            var source = documents.read(r.sourceVersionId());
            if (!admin && !actor.equals(source.createdBy())) throw new AccessDeniedException("Only the owner can create a new version");
            if (!source.sportId().equals(r.sportId())) throw new org.springframework.web.server.ResponseStatusException(
                    org.springframework.http.HttpStatus.BAD_REQUEST, "Sport cannot change across versions");
        }
        UUID ruleId = r.protocol() == null ? null : protocols.publish(r.sportId(), r.name().trim(), r.protocol(), actor);
        var document = documents.publish(r.sourceVersionId(), r.sportId(), r.code(), r.name(), r.description(),
                r.versionNo(), r.effectiveFrom(), r.notes(), r.conditions(), ruleId, actor, admin);
        return new Response(document, r.protocol());
    }
    @GetMapping("/{id}") @Transactional(readOnly = true)
    public Response get(@PathVariable UUID id, @AuthenticationPrincipal AuthenticatedUserPrincipal principal, Authentication authentication) {
        var document = documents.read(id);
        if (!admin(authentication) && !actor(principal).equals(document.createdBy())) {
            throw new AccessDeniedException("Only the owner can open this version for revision");
        }
        return new Response(document, document.binding() == null ? null : protocols.read(document.binding().ruleVersionId()));
    }
    private UUID actor(AuthenticatedUserPrincipal p) {
        if (p == null || p.getUserId() == null) throw new AccessDeniedException("Authenticated principal is required");
        return p.getUserId();
    }
    private boolean admin(Authentication a) {
        return a != null && a.getAuthorities().stream().anyMatch(v -> "ROLE_PLATFORM_ADMIN".equals(v.getAuthority()));
    }
}

    private ResolvedProtocolRules legacy(JsonNode node) {
        JsonNode capacityNode = node.path("unitCapacity");
        if (!capacityNode.isIntegralNumber() || !capacityNode.canConvertToInt()
                || capacityNode.intValue() < 1 || capacityNode.intValue() > 1024) {
            throw error("protocol.invalid_settings", "unitCapacity must be an integer in 1..1024");
        }
        int capacity = capacityNode.intValue();
        String key = node.path("seedMetricKey").asText(null), prefix = node.path("unitLabelPrefix").asText("Heat");
        if (key == null || key.isBlank() || prefix.isBlank() || prefix.length() > 80) {
            throw error("protocol.invalid_settings", "Invalid seedMetricKey or unitLabelPrefix");
        }
        List<Integer> allowed = IntStream.rangeClosed(1, capacity).boxed().toList();
        List<Integer> order = integers(node.path("positionOrder"));
        if (order.size() != capacity || new HashSet<>(order).size() != capacity
                || !new HashSet<>(order).equals(new HashSet<>(allowed))) throw error(
                        "protocol.invalid_position_order", "Invalid legacy positionOrder");
        List<RuleBlockRequest> blocks = List.of(
                new RuleBlockRequest(0, "ELIGIBILITY_CONFIRMED_ONLY", json.createObjectNode()),
                new RuleBlockRequest(1, "SEED_BY_METRIC_ASC_NULLS_LAST", json.createObjectNode()),
                new RuleBlockRequest(2, "GROUP_BY_CAPACITY", json.createObjectNode()),
                new RuleBlockRequest(3, "ASSIGN_CENTER_OUT", json.createObjectNode()));
        return new ResolvedProtocolRules(null, capacity, prefix, key, order, allowed,
                "FASTEST_FIRST", "LAST", blocks);
    }
    private List<Integer> integers(JsonNode array) {
        if (!array.isArray()) throw error("protocol.invalid_settings", "Positions must be an array");
        List<Integer> result = new ArrayList<>();

            if (!value.isIntegralNumber() || !value.canConvertToInt()) throw error(
                    "protocol.invalid_settings", "Positions must be integers");
            result.add(value.intValue());
        }
        return result;
    }

        if (raw == null || raw.isBlank()) return json.createObjectNode();
        JsonNode node;
        try { node = json.readTree(raw); }
        catch (Exception ex) { throw error(errorCode, "Invalid JSON"); }
        if (node == null || !node.isObject()) throw error(errorCode, "Expected a JSON object");
        return node;

import java.math.BigDecimal;
import java.util.*;
import com.acme.sportplatform.common.exception.BusinessException;
import com.acme.sportplatform.protocol.RegistrationProtocolAccess.ConfirmedRegistrationView;
import com.acme.sportplatform.protocol.api.RuleBlockRequest;

import com.fasterxml.jackson.databind.ObjectMapper;

/** Executes existing rule blocks and validates assignments, without persistence or authentication. */
public final class PrepareStartProtocol {
    private final RuleExecutorRegistry registry;
    private final ObjectMapper json;
    public PrepareStartProtocol(RuleExecutorRegistry registry, ObjectMapper json) {
        this.registry = Objects.requireNonNull(registry); this.json = Objects.requireNonNull(json);
    }
    public PreparedStartProtocol prepare(UUID disciplineId, ResolvedProtocolRules rules,
                                          List<ConfirmedRegistrationView> confirmed) {
        Objects.requireNonNull(disciplineId); Objects.requireNonNull(rules);
        confirmed = List.copyOf(confirmed);
        if (confirmed.isEmpty()) throw error("protocol.no_confirmed_registrations",
                "No confirmed registrations found; existing protocol has not been replaced");
        var eligibility=rules.blocks().stream().filter(block -> "ELIGIBILITY_CONFIRMED_ONLY".equals(block.executorCode())).toList();
        if(eligibility.size()>1) throw error("protocol.invalid_rule_pipeline", "Duplicate eligibility stage");
        if(!eligibility.isEmpty()) new ProtocolEligibilityVerifier(json).verify(confirmed,
                ProtocolEligibilityPolicy.parse(eligibility.get(0).config()));
        List<SeedParticipant> participants = confirmed.stream().map(reg -> new SeedParticipant(
                reg.registrationId(), metric(reg.registrationMeta(), rules.metricKey()))).toList();
        Map<UUID, BigDecimal> metrics = new HashMap<>();
        for (SeedParticipant participant : participants) {
            if (participant.registrationId() == null || metrics.containsKey(participant.registrationId())) {
                throw error("protocol.invalid_participants", "Registration IDs must be present and unique");
            }
            metrics.put(participant.registrationId(), participant.seedMetric());
        }
        var grouping = rules.blocks().stream().filter(b -> "GROUP_BY_CAPACITY".equals(b.executorCode())).toList();
        if (grouping.size() > 1) throw error("protocol.invalid_rule_pipeline", "Duplicate grouping stage");
        var categoryPolicy = ProtocolCategoryPolicy.parse(grouping.isEmpty() ? json.createObjectNode() : grouping.get(0).config());
        var membership = categoryPolicy.classify(confirmed, json);
        List<List<SeedParticipant>> batches = new ArrayList<>();
        if (categoryPolicy.enabled()) {
            for (String group : categoryPolicy.groupOrder()) {
                var batch = participants.stream().filter(p -> group.equals(membership.get(p.registrationId()).seedingGroupCode())).toList();
                if (!batch.isEmpty()) batches.add(batch);
            }
        } else batches.add(participants);
        List<GeneratedCompetitionUnit> units = new ArrayList<>();
        for (var batch : batches) {
            var context = new ProtocolGenerationContext(disciplineId, rules.capacity(), rules.prefix(),
                    rules.order(), rules.allowed(), rules.heatOrder(), rules.partialHeat(), batch);
            for (RuleBlockRequest block : rules.blocks()) registry.get(block.executorCode()).apply(context);
            validateOutput(context, batch.stream().map(SeedParticipant::registrationId).collect(java.util.stream.Collectors.toSet()));
            for (var group : context.getGroupedAssignments()) {
                var assignments = group.stream().map(a -> {
                    var category = membership.get(a.registrationId());
                    return new GeneratedAssignment(a.registrationId(), a.laneOrPosition(), metrics.get(a.registrationId()),
                            category == null ? null : category.categoryCode(), category == null ? null : category.seedingGroupCode());
                }).toList();
                int sequence = units.size() + 1;
                units.add(new GeneratedCompetitionUnit(UUID.randomUUID(), rules.prefix() + " " + sequence, sequence, assignments));
            }
        }
        return new PreparedStartProtocol(disciplineId, participants, units);
    }
    private JsonNode object(String raw, String errorCode) {
        if (raw == null || raw.isBlank()) return json.createObjectNode();
        JsonNode node;
        try { node = json.readTree(raw); }
        catch (Exception ex) { throw error(errorCode, "Invalid JSON"); }
        if (node == null || !node.isObject()) throw error(errorCode, "Expected a JSON object");
        return node;
    }
    private BigDecimal metric(String raw, String key) {
        JsonNode value = object(raw, "protocol.invalid_seed_metric").path(key);
        if (value.isMissingNode() || value.isNull()) return null;
        if (!value.isNumber()) throw error("protocol.invalid_seed_metric", "Seed metric must be numeric");
        return value.decimalValue();
    }
    private void validateOutput(ProtocolGenerationContext context, Set<UUID> expected) {
        Set<UUID> actual = new HashSet<>(); Set<Integer> allowed = new HashSet<>(context.getAllowedPositions());
        for (var group : context.getGroupedAssignments()) {
            if (group.isEmpty() || group.size() > context.getUnitCapacity()) throw invalidOutput();
            Set<Integer> occupied = new HashSet<>();
            for (int i = 0; i < group.size(); i++) {
                var a = group.get(i);
                if (!expected.contains(a.registrationId()) || !actual.add(a.registrationId())
                        || !allowed.contains(a.laneOrPosition()) || !occupied.add(a.laneOrPosition())
                        || !Objects.equals(a.laneOrPosition(), context.getPositionOrder().get(i))) throw invalidOutput();
            }
        }
        if (!actual.equals(expected)) throw invalidOutput();
    }
    private BusinessException invalidOutput() {
        return error("protocol.invalid_generated_assignments", "Generated assignments violate invariants");
    }
    private BusinessException error(String code, String message) { return new BusinessException(code, message); }
}

package com.acme.sportplatform.protocol.application.strategy;
import java.util.*;
import org.springframework.stereotype.Component;
import com.acme.sportplatform.protocol.application.*;
import com.acme.sportplatform.common.exception.BusinessException;
@Component
public class GroupByCapacityExecutor implements RuleBlockExecutor {
    @Override public String code() { return "GROUP_BY_CAPACITY"; }
    @Override public void apply(ProtocolGenerationContext context) {
        int capacity = context.getUnitCapacity();
        if (capacity < 1) throw new BusinessException("protocol.invalid_settings", "Invalid capacity");
        List<SeedParticipant> participants = context.getParticipants();
        context.getGroupedAssignments().clear();
        boolean slowestFirst = "SLOWEST_FIRST".equals(context.getHeatOrder());
        boolean partialFirstInOutput = "FIRST".equals(context.getPartialHeat());
        boolean partialFirstBeforeReverse = slowestFirst ? !partialFirstInOutput : partialFirstInOutput;
        int remainder = participants.size() % capacity;
        int offset = 0;
        while (offset < participants.size()) {
            int size = offset == 0 && partialFirstBeforeReverse && remainder > 0
                    ? remainder : Math.min(capacity, participants.size() - offset);
            List<GeneratedAssignment> group = new ArrayList<>();
            for (SeedParticipant participant : participants.subList(offset, offset + size)) {
                group.add(new GeneratedAssignment(participant.registrationId(), null, participant.seedMetric()));
            }
            context.getGroupedAssignments().add(group);
            offset += size;
        }
        if (slowestFirst) Collections.reverse(context.getGroupedAssignments());
    }
}

package com.acme.sportplatform.regulations.application;

import java.time.LocalDate;
import java.util.List;
import java.util.UUID;
import jakarta.persistence.EntityManager;
import jakarta.persistence.LockModeType;
import jakarta.persistence.PersistenceContext;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import com.acme.sportplatform.regulations.RegulationBundleAccess;
import com.acme.sportplatform.regulations.api.*;
import com.acme.sportplatform.regulations.infrastructure.jpa.*;

@Service
public class RegulationBundleService implements RegulationBundleAccess {
    private final RegulationService service;
    private final RegulationVersionRepository versions;
    private final JdbcTemplate jdbc;
    @PersistenceContext private EntityManager em;
    public RegulationBundleService(RegulationService service, RegulationVersionRepository versions, JdbcTemplate jdbc) {
        this.service = service; this.versions = versions; this.jdbc = jdbc;
    }
    @Override @Transactional(readOnly = true)
    public Binding binding(UUID id) {
        return jdbc.query("""
                select protocol_rule_set_version_id, competition_format, unit_type
                from regulation_protocol_bindings where regulation_version_id = ?
                """, (rs, row) -> new Binding(rs.getObject(1, UUID.class), rs.getString(2), rs.getString(3)), id)
                .stream().findFirst().orElse(null);
    }
    @Override @Transactional(readOnly = true)
    public Document read(UUID id) {
        var v = service.getVersion(id);
        var t = service.getTemplate(v.regulationTemplateId());
        var sections = v.rulesJson() == null ? null : v.rulesJson().sections();
        var conditions = sections == null ? List.<String>of()
                : sections.stream().map(RegulationSection::title).toList();
        return new Document(v.id(), t.id(), t.sportId(), t.code(), t.name(), t.description(),
                v.versionNo(), v.status(), v.effectiveFrom(), v.notes(), v.createdBy(), conditions, binding(id));
    }
    @Override @Transactional
    public Document publish(UUID sourceId, UUID sportId, String code, String name,
            String description, int versionNo, LocalDate effectiveFrom, String notes,
            List<String> conditions, UUID ruleVersionId, UUID actorId, boolean admin) {
        UUID templateId;
        if (sourceId == null) {
            if (versionNo != 1) throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "A new regulation starts at version 1");
            templateId = service.createTemplate(new CreateRegulationTemplateRequest(
                    sportId, code.trim().toUpperCase(java.util.Locale.ROOT), name.trim(), description)).id();
        } else {
            var source = read(sourceId);
            if (!admin && !actorId.equals(source.createdBy())) {
                throw new org.springframework.security.access.AccessDeniedException("Only the owner can publish a new version");
            }
            if (!source.sportId().equals(sportId)) throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Sport cannot change across versions");
            templateId = source.templateId();
            var template = em.find(RegulationTemplateEntity.class, templateId);
            em.refresh(template, LockModeType.PESSIMISTIC_WRITE);
            Integer latest = em.createQuery("select max(v.versionNo) from RegulationVersionEntity v where v.regulationTemplateId = :id", Integer.class)
                    .setParameter("id", templateId).getSingleResult();
            if (latest != null && versionNo <= latest) throw new ResponseStatusException(HttpStatus.CONFLICT, "Use a version number greater than all existing versions");
        }
        var sections = java.util.stream.IntStream.range(0, conditions.size())
                .mapToObj(i -> new RegulationSection("condition-" + (i + 1), conditions.get(i).trim())).toList();
        var version = service.createVersion(templateId, new CreateRegulationVersionRequest(
                versionNo, new RegulationRules(sections), notes, effectiveFrom));
        var entity = versions.findById(version.id()).orElseThrow();
        entity.setCreatedBy(actorId);
        versions.flush();
        if (ruleVersionId != null) {
            jdbc.update("""
                    insert into regulation_protocol_bindings
                    (regulation_version_id, protocol_rule_set_version_id, competition_format, unit_type, created_by_user_id)
                    values (?, ?, 'INDIVIDUAL', 'HEAT', ?)
                    """, version.id(), ruleVersionId, actorId);
        }
        service.publishVersion(version.id());
        return read(version.id());
    }
}

import com.acme.sportplatform.regulations.infrastructure.jpa.RegulationVersionRepository;
import com.acme.sportplatform.sports.SportLookup;

@Service
public class RegulationService {

    private final RegulationTemplateRepository regulationTemplateRepository;
    private final RegulationVersionRepository regulationVersionRepository;
    private final SportLookup sportLookup;

    public RegulationService(
            RegulationTemplateRepository regulationTemplateRepository,
            RegulationVersionRepository regulationVersionRepository,
            SportLookup sportLookup
    ) {
        this.regulationTemplateRepository = regulationTemplateRepository;
        this.regulationVersionRepository = regulationVersionRepository;
        this.sportLookup = sportLookup;
    }

    @Transactional(readOnly = true)
    public List<RegulationTemplateResponse> getTemplates() {
        return regulationTemplateRepository.findAll().stream()
                .map(this::mapTemplate)
                .toList();
    }

    @Transactional
    public RegulationTemplateResponse createTemplate(CreateRegulationTemplateRequest request) {
        if (!sportLookup.existsById(request.sportId())) {
            throw new BusinessException("sports.sport_not_found", "Sport not found");
        }

        if (regulationTemplateRepository.existsBySportIdAndCode(request.sportId(), request.code())) {
            throw new BusinessException(
                    "regulations.template_already_exists",
                    "Regulation template already exists"
            );
        }

        RegulationTemplateEntity entity = new RegulationTemplateEntity();
        entity.setId(UUID.randomUUID());
        entity.setSportId(request.sportId());
        entity.setCode(request.code());
        entity.setName(request.name());
        entity.setDescription(request.description());
        entity.setActive(true);
        entity.setCreatedAt(OffsetDateTime.now(ZoneOffset.UTC));

        return mapTemplate(regulationTemplateRepository.save(entity));
    }

    @Transactional(readOnly = true)
    public RegulationTemplateResponse getTemplate(UUID id) {
        return mapTemplate(regulationTemplateRepository.findById(id)
                .orElseThrow(() -> new BusinessException(
                        "regulations.template_not_found",
                        "Regulation template not found"
                )));
    }

    @Transactional
    public RegulationVersionResponse createVersion(UUID templateId, CreateRegulationVersionRequest request) {
        RegulationTemplateEntity template = regulationTemplateRepository.findById(templateId)
                .orElseThrow(() -> new BusinessException(
                        "regulations.template_not_found",
                        "Regulation template not found"
                ));

        if (regulationVersionRepository.existsByRegulationTemplateIdAndVersionNo(templateId, request.versionNo())) {
            throw new BusinessException(
                    "regulations.version_already_exists",
                    "Regulation version already exists"
            );
        }

        RegulationVersionEntity entity = new RegulationVersionEntity();
        entity.setId(UUID.randomUUID());
        entity.setRegulationTemplateId(template.getId());
        entity.setVersionNo(request.versionNo());
        entity.setStatus("draft");
        entity.setEffectiveFrom(request.effectiveFrom());
        entity.setRulesJson(request.rulesJson());
        entity.setNotes(request.notes());
        entity.setCreatedAt(OffsetDateTime.now(ZoneOffset.UTC));
        entity.setCreatedBy(null);

        return mapVersion(regulationVersionRepository.save(entity));
    }

    @Transactional(readOnly = true)
    public RegulationVersionResponse getVersion(UUID id) {
        return mapVersion(regulationVersionRepository.findById(id)
                .orElseThrow(() -> new BusinessException(
                        "regulations.version_not_found",
                        "Regulation version not found"
                )));
    }

    @Transactional
    public RegulationVersionResponse publishVersion(UUID id) {
        RegulationVersionEntity entity = regulationVersionRepository.findById(id)
                .orElseThrow(() -> new BusinessException(
                        "regulations.version_not_found",
                        "Regulation version not found"
                ));

        if (!"draft".equals(entity.getStatus())) {
            throw new BusinessException(
                    "regulations.invalid_status_transition",
                    "Only draft regulation version can be published"
            );
        }

        entity.setStatus("published");
        return mapVersion(regulationVersionRepository.save(entity));
    }

    @Transactional(readOnly = true)
        public List<RegulationVersionResponse> getVersions() {
        return regulationVersionRepository.findAll().stream()
                .map(this::mapVersion)
                .toList();
        }

        return new RegulationTemplateResponse(
                entity.getId(),
                entity.getSportId(),
                entity.getCode(),
                entity.getName(),
                entity.getDescription(),
                entity.isActive(),
                entity.getCreatedAt()
        );
    }

    private RegulationVersionResponse mapVersion(RegulationVersionEntity entity) {
        return new RegulationVersionResponse(
                entity.getId(),
                entity.getRegulationTemplateId(),
                entity.getVersionNo(),
                entity.getStatus(),
                entity.getEffectiveFrom(),
                entity.getRulesJson(),
                entity.getNotes(),
                entity.getCreatedBy(),
                entity.getCreatedAt()
        );
    }
