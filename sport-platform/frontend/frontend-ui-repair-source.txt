import { readApiFailure } from '../lib/standings-ui.mjs'
import type { FieldViolation } from '../lib/standings-ui.mjs'

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL

export class ApiError extends Error {
  public readonly status: number
  public readonly code?: string
  public readonly violations: FieldViolation[]

  constructor(
    status: number,
    message: string,
    code?: string,
    violations: FieldViolation[] = [],
  ) {
    super(message)

    this.name = 'ApiError'
    this.status = status
    this.code = code
    this.violations = violations
  }
}

type ApiFetchOptions = RequestInit & {
  accessToken?: string | null
}

export async function apiFetch<T>(
  path: string,
  options: ApiFetchOptions = {},
): Promise<T> {
  if (!API_BASE_URL) {
    throw new Error(
      'VITE_API_BASE_URL is not configured. Create frontend/.env.local and restart Vite.',
    )
  }

  const { accessToken, ...requestOptions } = options
  const headers = new Headers(requestOptions.headers)

  headers.set('Accept', 'application/json')

  if (accessToken) {
    headers.set('Authorization', `Bearer ${accessToken}`)
  }

  if (
    typeof requestOptions.body === 'string' &&
    !headers.has('Content-Type')
  ) {
    headers.set('Content-Type', 'application/json')
  }

  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...requestOptions,
    headers,
  })

  if (!response.ok) {
    let payload: unknown

    try {
      payload = await response.json()
    } catch {
      payload = undefined
    }

    const failure = readApiFailure(payload, response.status)
    throw new ApiError(response.status, failure.message, failure.code, failure.violations)
  }

  if (response.status === 204) {
    return undefined as T
  }

  const contentType = response.headers.get('content-type')

  if (!contentType?.includes('application/json')) {
    return undefined as T
  }

  return (await response.json()) as T
}

function checkedStandings(data: Standings): Standings {
  if (!data || !Array.isArray(data.entries) || typeof data.eventId !== 'string' || typeof data.eventDisciplineId !== 'string')
    throw new Error('Сервер вернул неожиданный формат зачёта.')
  return data
}
function checkedFinal(data: FinalStandings): FinalStandings {
  if (!data || typeof data.id !== 'string' || !Number.isInteger(data.revision) || !['CONFIRMED','REVOKED'].includes(data.status))
    throw new Error('Сервер вернул неожиданный формат итогового снимка.')
  checkedStandings(data.standings)
  return data
}
export async function getStandings(eventId:string,disciplineId:string):Promise<Standings> {
  return checkedStandings(await apiFetch<Standings>(`/api/v1/public/${scope(eventId,disciplineId)}/standings`))
}
export async function getFinalStandings(eventId:string,disciplineId:string):Promise<FinalStandings> {
  const data=checkedFinal(await apiFetch<FinalStandings>(`/api/v1/public/${scope(eventId,disciplineId)}/final-standings`))
  if(data.status!=='CONFIRMED') throw new Error('Публичный endpoint вернул неактивную ревизию. Обновите данные.')
  return data
}

function getErrorMessage(error: unknown, fallback: string): string {
  if (error instanceof ApiError) {
    if (error.status === 403) {
      return 'Нет доступа к управлению мероприятием.'
    }

    return error.message || fallback
  }

  return fallback
}

function statusClassName(status: string): string {
  switch (getEventStatusTone(status)) {
    case 'draft':
      return `${styles.statusBadge} ${styles.statusDraft}`
    case 'published':

    case 'open':
      return `${styles.statusBadge} ${styles.statusOpen}`
    case 'closed':
      return `${styles.statusBadge} ${styles.statusClosed}`
    case 'live':
      return `${styles.statusBadge} ${styles.statusLive}`
    case 'completed':
      return `${styles.statusBadge} ${styles.statusCompleted}`
    case 'archived':
      return `${styles.statusBadge} ${styles.statusArchived}`
    default:
      return `${styles.statusBadge} ${styles.statusNeutral}`

function formatDate(value?: string | null): string {
  if (!value) {
    return '—'
  }

  const date = new Date(value)

  if (Number.isNaN(date.getTime())) {
    return '—'
  }

  return new Intl.DateTimeFormat('ru-RU', {
    dateStyle: 'medium',
    timeStyle: 'short',
  }).format(date)
}

export function OrganizerEventManagePage() {
  const { eventId } = useParams()
  const { session } = useAuth()
  const accessToken = session?.accessToken

  const [event, setEvent] = useState<Event | null>(null)
  const [disciplines, setDisciplines] = useState<EventDiscipline[]>([])
  const [isLoading, setIsLoading] = useState(true)
  const [errorMessage, setErrorMessage] = useState<string | null>(null)
  const [actionMessage, setActionMessage] = useState<string | null>(null)
  const [publishingId, setPublishingId] = useState<string | null>(null)
  const [eventAction, setEventAction] = useState<EventAction | null>(null)

  useEffect(() => {
    if (!eventId || !accessToken) {
      setIsLoading(false)
      return
    }

    let isMounted = true

    async function loadData() {
      setIsLoading(true)
      setErrorMessage(null)

      try {
        const [eventData, disciplinesData] = await Promise.all([
          getEventById(eventId, accessToken),
          getEventDisciplines(eventId, accessToken),
        ])

        if (!isMounted) {
          return
        }

        setEvent(eventData)
        setDisciplines(disciplinesData)
      } catch (error) {
        if (isMounted) {
          setErrorMessage(
            getErrorMessage(error, 'Не удалось загрузить мероприятие.'),
          )
        }
      } finally {
        if (isMounted) {
          setIsLoading(false)
        }
      }
    }

    void loadData()

    return () => {
      isMounted = false
    }
  }, [eventId, accessToken])

  async function handlePublishDiscipline(disciplineId: string) {
    if (!eventId || !accessToken) {
      return
    }

    setPublishingId(disciplineId)
    setErrorMessage(null)
    setActionMessage(null)

    try {
      const updatedDiscipline = await publishEventDiscipline(
        eventId,
        disciplineId,
        accessToken,
      )

      setDisciplines((current) =>
        current.map((discipline) =>
          discipline.id === disciplineId ? updatedDiscipline : discipline,
        ),
      )

      setActionMessage('Дисциплина опубликована.')
    } catch (error) {
      setErrorMessage(
        getErrorMessage(error, 'Не удалось опубликовать дисциплину.'),
      )
    } finally {
      setPublishingId(null)
    }
  }

  async function handleEventAction(action: EventAction) {
    if (!eventId || !accessToken || !event) {
      return
    }

    setEventAction(action)
    setErrorMessage(null)
    setActionMessage(null)

    try {
      let updatedEvent: Event
      let successMessage: string

      switch (action) {
        case 'publish':
          updatedEvent = await publishEvent(eventId, accessToken)
          successMessage = 'Мероприятие опубликовано.'
          break

        case 'open-registration':
          updatedEvent = await openEventRegistration(eventId, accessToken)
          successMessage = 'Регистрация открыта.'
          break

        case 'close-registration':
          updatedEvent = await closeEventRegistration(eventId, accessToken)
          successMessage = 'Регистрация закрыта.'
          break

        case 'complete':
          updatedEvent = await completeEvent(eventId, accessToken)
          successMessage = 'Мероприятие завершено.'
          break
      }

      setEvent(updatedEvent)
      setActionMessage(successMessage)
    } catch (error) {
      setErrorMessage(
        getErrorMessage(error, 'Не удалось изменить статус мероприятия.'),
      )
    } finally {
      setEventAction(null)
    }
  }

  const eventStatus = normalizeEventStatus(event?.status)

  return (
    <section className={styles.page}>
      <NavLink className={styles.backLink} to="/organizer/events">

      ) : null}

      {errorMessage ? (

        </p>

function getErrorMessage(error: unknown, fallback: string) {
  if (error instanceof ApiError) {
    if (error.status === 403) {
      return 'У вашей учётной записи нет прав для выполнения этого действия.'
    }

    if (error.status === 404) {
      return 'Соревновательная единица не найдена.'
    }

    return error.message
  }

  return fallback
}

function createEntryDrafts(
  entries: CompetitionUnitEntryView[],
): Record<string, EntryDraft> {
  return Object.fromEntries(
    entries.map((entry) => [
      entry.entryId,
      {
        rawValue: entry.rawValue ?? '',
        status: isEntryResultStatus(entry.resultStatus)
          ? entry.resultStatus
          : 'PENDING',
      },
    ]),
  )
}

function isEntryResultStatus(value: string | null): value is EntryResultStatus {
  return (
    value === 'PENDING' ||
    value === 'DID_NOT_START' ||
    value === 'DID_NOT_FINISH' ||
    value === 'DISQUALIFIED'
  )
}

function ProtocolPageContent({ eventId, disciplineId, accessToken }: { eventId: string; disciplineId: string; accessToken: string }) {
  const [data, setData] = useState<{ event: Event; discipline: EventDiscipline } | null>(null)
  const [error, setError] = useState<unknown>(null)
  const [loading, setLoading] = useState(true)
  const [epoch, setEpoch] = useState(0)
  useEffect(() => {
    let valid = true
    setLoading(true); setData(null); setError(null)
    void Promise.all([getEventById(eventId, accessToken), getEventDisciplines(eventId, accessToken)])
      .then(([event, disciplines]) => {
        if (!valid) return
        const discipline = disciplines.find(item => item.id === disciplineId)
        if (!discipline) throw new Error('Дисциплина не найдена в этом мероприятии.')
        setData({ event, discipline })
      })
      .catch(failure => { if (valid) setError(failure) })
      .finally(() => { if (valid) setLoading(false) })
    return () => { valid = false }
  }, [eventId, disciplineId, accessToken, epoch])
  return <section className={styles.page}>
    <Link className={styles.link} to={`/organizer/events/${encodeURIComponent(eventId)}`}>← К мероприятию</Link>
    <h1>Зачёт и итоговый протокол</h1>
    {loading ? <p role="status">Загружаем дисциплину…</p> : null}
    {error ? <div className={styles.error} role="alert"><p>{protocolErrorMessage(error)}</p>
      <button type="button" className={styles.button} onClick={() => setEpoch(value => value + 1)}>Повторить загрузку</button></div> : null}
    {!loading && !error && data ? <>
      <p>{data.event.title} — {data.discipline.name}</p>
      <div className={styles.toolbar}>
        {data.event.publicSlug ? <Link className={styles.link} to={`/events/${encodeURIComponent(data.event.publicSlug)}/disciplines/${encodeURIComponent(disciplineId)}/standings`}>Публичный зачёт</Link> : null}
      </div>
      <DisciplineProtocolBoard eventId={eventId} disciplineId={disciplineId} accessToken={accessToken} />
    </> : null}
  </section>
}
export function OrganizerDisciplineProtocolPage() {
  const { eventId, disciplineId } = useParams<{ eventId: string; disciplineId: string }>()

  if (!eventId || !disciplineId || !session?.accessToken) return <p role="alert">Не указана дисциплина или отсутствует авторизованная сессия.</p>

}

  }
  return messages[error?.code] || (typeof error?.message === 'string' ? error.message : 'Не удалось выполнить запрос. Проверьте соединение и обновите данные.')

export function createConfirmationAttempt(note, programmeComplete, createId = () => globalThis.crypto.randomUUID()) {
  if (programmeComplete !== true) throw new Error('Явно подтвердите завершённость всей программы дисциплины.')
  const normalized = note.trim()
  if (normalized.length > 500) throw new Error('Примечание не должно превышать 500 символов.')
  let requestId
  try { requestId = createId() } catch { throw new Error('Не удалось создать безопасный requestId. Откройте интерфейс через HTTPS или localhost.') }
  if (typeof requestId !== 'string' || !/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(requestId)) {
    throw new Error('Не удалось создать корректный requestId.')
  }
  return Object.freeze({ requestId, programmeComplete: true, note: normalized })
}
export function createRevocationAttempt(snapshot, reason) {
  const normalized = reason.trim()
  if (!normalized || normalized.length > 1000) throw new Error('Укажите причину снятия: от 1 до 1000 символов.')
  return Object.freeze({ id: snapshot.id, revision: snapshot.revision, reason: normalized })
}

  }

  }

}

import { useEffect, useState } from 'react'
import {
  getDisciplineStandings, getFinalStandings, getFinalizationHistory,
  type DisciplineStandings, type FinalSnapshot,
} from '../../api/standings.api'
import { getCompetitionUnitsByDiscipline, type CompetitionUnitResponse } from '../../api/results.api'

type Resource = 'current' | 'active' | 'history' | 'units'
type ProtocolData = {
  loading: boolean
  current: DisciplineStandings | null
  active: FinalSnapshot | null
  history: FinalSnapshot[] | null
  units: CompetitionUnitResponse[] | null
  errors: Partial<Record<Resource, unknown>>
}
const emptyData = (): ProtocolData => ({ loading: true, current: null, active: null, history: null, units: null, errors: {} })

export function useProtocolData(eventId: string, disciplineId: string, accessToken?: string) {
  const [data, setData] = useState<ProtocolData>(emptyData)
  const [epoch, setEpoch] = useState(0)
  useEffect(() => {
    let valid = true
    const controller = new AbortController()
    setData(emptyData())
    async function load() {
      const [current, active, history, units] = await Promise.allSettled([
        getDisciplineStandings(eventId, disciplineId, controller.signal),
        getFinalStandings(eventId, disciplineId, controller.signal),
        accessToken ? getFinalizationHistory(eventId, disciplineId, accessToken, controller.signal) : Promise.resolve(null),
        accessToken ? getCompetitionUnitsByDiscipline(disciplineId, accessToken) : Promise.resolve(null),
      ])
      if (!valid) return
      const errors: ProtocolData['errors'] = {}
      if (current.status === 'rejected') errors.current = current.reason
      if (active.status === 'rejected') errors.active = active.reason
      if (history.status === 'rejected') errors.history = history.reason
      if (units.status === 'rejected') errors.units = units.reason
      setData({
        loading: false, errors,
        current: current.status === 'fulfilled' ? current.value : null,
        active: active.status === 'fulfilled' ? active.value : null,
        history: history.status === 'fulfilled' ? history.value : null,
        units: units.status === 'fulfilled' ? units.value : null,
      })
    }
    void load()
    return () => { valid = false; controller.abort() }
  }, [eventId, disciplineId, accessToken, epoch])
  function refresh() {
    setData(current => ({ ...current, loading: true }))
    setEpoch(current => current + 1)
  }
  return { data, refresh }
}
